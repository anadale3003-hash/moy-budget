function paymentFunding(target,amount,date){
 const available=target==='mortgage'?regularMortgageAvailable(date):regularCreditAvailable(date);
 const reserve=Math.max(0,Math.round(available*100)/100);
 const reserveUsed=Math.min(amount,reserve);
 const extraAmount=Math.max(0,Math.round((amount-reserveUsed)*100)/100);
 return {reserveUsed,extraAmount};
}

function confirmRegularPayment(target,amount,date,extraSource='',part=''){
 if(!data.startBalances)return 'Сначала укажи стартовые остатки в настройках.';
 if(!Number.isFinite(amount)||amount<=0||!fromISO(date)||date>iso(new Date())||date<data.startBalances.date)return 'Проверь сумму и дату. Платёж должен быть после начала учёта и не позже сегодня.';
 if(target==='credit'){
  const outstanding=Math.max(0,num(data.debt)-repaymentTotal('credit')-regularPaymentSum('credit'));
  if(amount>outstanding+0.001)return 'Платёж превышает оставшийся учтённый долг: '+rub(outstanding);
 }
 const {reserveUsed,extraAmount}=paymentFunding(target,amount,date);
 if(extraAmount>0){
  if(!cats.some(c=>c.id===extraSource))return 'Выбери конверт, из которого списать дополнительные '+rub(extraAmount)+'.';
  const categoryBalance=categoryBalanceAt(extraSource,date);
  if(categoryBalance===null||categoryBalance+0.001<extraAmount)return 'В выбранном конверте недостаточно денег на дату списания. Доступно: '+rub(Math.max(0,categoryBalance||0))+'.';
 }
 data.regularPayments.push({id:'payment-'+Date.now()+'-'+Math.random().toString(36).slice(2,8),
  month,target,part:part==='advance'||part==='salary'?part:'',amount,date,reserveUsed,extraSource:extraAmount>0?extraSource:'',extraAmount});
 if(target==='credit')data.reserved=Math.max(0,Math.round((num(data.reserved)-reserveUsed)*100)/100);
 closePaymentEntry();renderAll();
 return null;
}

/* Regular bank payment confirmation now uses the shared reserve. Wage
   splits remain in allocation() for planning, not as a payment choice. */
function suggestedRegularPayment(target,date){
 const already=regularPaidForMonth(target,month);
 const due=target==='mortgage'
  ?Math.max(0,num(monthData().mortgage)-already)
  :Math.max(0,num(data.debt)-repaymentTotal('credit')-regularPaymentSum('credit'));
 if(target==='mortgage'&&data.paidMortgageMonth===month&&already===0)return 0;
 const available=Math.max(0,target==='mortgage'?regularMortgageAvailable(date):regularCreditAvailable(date));
 // Default to what has actually been funded, capped by the remaining due.
 // Never propose an unfunded full mortgage installment from the envelope.
 // Actual bank debit can still be entered manually if it differs.
 return Math.round(Math.min(due,available)*100)/100;
}
