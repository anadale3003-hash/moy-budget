/* v1.5.32: automatic, conservative economy discovery from recorded expenses.
   Only comparable categories with expenses in both periods are used;
   growth elsewhere offsets reductions, and transfers never exceed recorded funds. */
function analyticsEconomy(){
 const selected=month,nowDay=iso(new Date());
 const [year,m]=selected.split('-').map(Number);
 const previous=iso(new Date(year,m-2,1)).slice(0,7);
 const isCurrent=selected===currentMonth;
 const cutoff=isCurrent?new Date().getDate():31;
 const previousCutoff=isCurrent?Math.min(cutoff,new Date(year,m-1,0).getDate()):31;
 const groups=[],sources=['family','utilities','other'];
 const match=(entry,key,day)=>entry.date?.startsWith(key)&&Number(entry.date.slice(8,10))<=day;
 for(const id of sources){
   const past=data.expenses.filter(e=>e.category===id&&match(e,previous,previousCutoff));
   const present=data.expenses.filter(e=>e.category===id&&match(e,selected,cutoff));
   // Missing current records may mean purchases have not yet been entered.
   if(!past.length||!present.length)continue;
   const pastTotal=past.reduce((a,e)=>a+num(e.amount),0);
   const presentTotal=present.reduce((a,e)=>a+num(e.amount),0);
   const signed=pastTotal-presentTotal;
   const saved=(data.transfers||[]).filter(t=>t.kind==='economy'&&t.to==='savings'&&t.from===id&&t.period===selected)
    .reduce((a,t)=>a+num(t.amount),0);
   const balance=categoryBalanceAt(id,nowDay);
   const pastTypes=new Map(),presentTypes=new Map();
   for(const e of past){const key=String(e.type||'Без вида').trim()||'Без вида';pastTypes.set(key,(pastTypes.get(key)||0)+num(e.amount))}
   for(const e of present){const key=String(e.type||'Без вида').trim()||'Без вида';presentTypes.set(key,(presentTypes.get(key)||0)+num(e.amount))}
   const types=[...pastTypes.entries()].map(([name,value])=>({
     name,delta:Math.max(0,Math.round((value-(presentTypes.get(name)||0))*100)/100)
   })).filter(t=>t.delta>0).sort((x,y)=>y.delta-x.delta);
   groups.push({id,name:cats.find(c=>c.id===id)?.name||id,past:pastTotal,present:presentTotal,
     signed,decline:Math.max(0,signed),saved,balance,types,allocated:0,max:0});
 }
 const grossReduction=Math.max(0,Math.round(groups.reduce((a,g)=>a+g.signed,0)*100)/100);
 // Distribute the actual net improvement, not a sum of isolated reductions.
 const sorted=[...groups].sort((a,b)=>b.decline-a.decline);
 let uncovered=grossReduction;
 for(const group of sorted){
   group.allocated=Math.min(uncovered,group.decline);
   uncovered=Math.max(0,Math.round((uncovered-group.allocated)*100)/100);
 }
 const previouslyTransferred=groups.reduce((a,g)=>a+g.saved,0);
 let transferLeft=Math.max(0,Math.round((grossReduction-previouslyTransferred)*100)/100);
 const todayMonth=isCurrent;
 for(const group of sorted){
   const categoryRemaining=Math.max(0,group.allocated-group.saved);
   group.max=todayMonth&&group.balance!==null
     ?Math.max(0,Math.floor(Math.min(transferLeft,categoryRemaining,group.balance)*100)/100):0;
   transferLeft=Math.max(0,Math.round((transferLeft-group.max)*100)/100);
 }
 // Assign explanatory type declines only up to their category's allocated net savings.
 const lines=[];
 for(const group of sorted){
   let cap=group.allocated;
   for(const t of group.types){
     const value=Math.min(cap,t.delta);
     if(value>0.009)lines.push({category:group.id,label:group.name+' · '+t.name,amount:value});
     cap=Math.max(0,Math.round((cap-value)*100)/100);
     if(cap<0.01)break;
   }
 }
 lines.sort((x,y)=>y.amount-x.amount);
 return {selected,previous,cutoff,isCurrent,groups,lines,total:grossReduction,
   available:groups.reduce((a,g)=>a+g.max,0),comparable:groups.length,
   previousSaved:previouslyTransferred};
}
function ladderEconomy(source){
 const auto=analyticsEconomy(),found=auto.groups.find(g=>g.id===source);
 if(!found)return {error:'Для этой категории пока недостаточно записей в обоих месяцах.',max:0};
 return {last:found.past,now:found.present,delta:found.allocated,saved:found.saved,
   remaining:Math.max(0,found.allocated-found.saved),balance:found.balance,max:found.max};
}

/* v1.5.35: capital goal planner. Same reserve accounting as the ladder above. */
function capitalSnapshot(goalOverride){
 const settings=ladderSettings(),reserve=actualReserve();
 const baseReserve=num(settings.reserve);
 const savings=categoryBalanceAt('savings',iso(new Date()));
 const booked=savings===null?0:Math.max(0,savings);
 // When the original cushion is already included in savings, subtract only that part.
 // New direct cushion income is held separately and must never be deducted twice.
 const extra=settings.included?Math.max(0,booked-baseReserve):booked;
 const total=reserve+extra;
 const automaticGoal=total<1000000?1000000:(Math.floor(total/500000)+1)*500000;
 const userGoal=typeof settings.targetAmount==='number'&&Number.isFinite(settings.targetAmount)
   &&settings.targetAmount>=reserve&&settings.targetAmount<=1000000000;
 const override=typeof goalOverride==='number'&&Number.isInteger(goalOverride)
   &&goalOverride>=reserve&&goalOverride<=1000000000;
 const goal=override?goalOverride:(userGoal?settings.targetAmount:automaticGoal);
 return {settings,reserve,savings,extra,total,goal,automatic:!override&&!userGoal,
   remaining:Math.max(0,goal-total)};
}
