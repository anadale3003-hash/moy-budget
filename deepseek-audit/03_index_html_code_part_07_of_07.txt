  const receipts=Array.isArray(d.receipts)?d.receipts.filter(x=>validDate(x.date)&&x.date<=today):[];
  const periods=[...new Set([...expenses.map(x=>x.date.slice(0,7)),...receipts.map(x=>x.date.slice(0,7)),...incomes.map(x=>x.date.slice(0,7))])].filter(x=>x<=month).sort().slice(-6);
  if(!periods.includes(month)) periods.push(month);
  const mortgage=amount(d.months?.[month]?.mortgage??d.mortgageMonthly??55000);
