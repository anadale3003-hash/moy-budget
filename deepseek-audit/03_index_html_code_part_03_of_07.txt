function nextPaydays(){

  // the credit deadline but associate both payments with the same plan month.

 const asOf=iso(new Date());
 const envelopes=cats.reduce((total,c)=>total+categoryBalanceAt(c.id,asOf),0);
 const mortgage=num(data.startBalances.mortgage)
   +data.receipts.filter(r=>r.date>=data.startBalances.date&&r.date<=asOf).reduce((a,r)=>a+num(r.amounts?.mortgage),0)
   +data.extraIncomes.filter(i=>i.destination==='mortgage'&&i.date>=data.startBalances.date&&i.date<=asOf).reduce((a,i)=>a+num(i.amount),0)
   -mortgageReserveSpent(data.startBalances.date,asOf);
 // data.reserved includes manual credit-card reserves and reserves from received wages.
 // Exclude future-dated receipt allocations if any exist.
 const futureCredit=data.receipts.filter(r=>r.date>asOf).reduce((a,r)=>a+num(r.amounts?.credit),0);
 const creditReserve=num(data.reserved)-futureCredit
   +data.extraIncomes.filter(i=>i.destination==='credit'&&i.date>=data.startBalances.date&&i.date<=asOf).reduce((a,i)=>a+num(i.amount),0);
 const savings=categoryBalanceAt('savings',asOf);
 const available=envelopes-savings;
 const reserve=mortgage+creditReserve;
 return {envelopes,savings,available,reserve,mortgage,creditReserve,total:envelopes+reserve};
}

function regularCreditAvailable(date){
 if(!data.startBalances)return 0;
 const futureCredit=(data.receipts||[]).filter(r=>r.date>date).reduce((a,r)=>a+num(r.amounts?.credit),0);
 const extraFuture=(data.extraIncomes||[]).filter(x=>x.destination==='credit'&&x.date>date).reduce((a,x)=>a+num(x.amount),0);
 // data.reserved already excludes confirmed credit-card payments.
 return num(data.reserved)-futureCredit+incomeAmountFor('credit',data.startBalances.date)-extraFuture+(data.regularPayments||[]).filter(p=>p.target==='credit'&&p.date>date).reduce((a,p)=>a+num(p.amount),0);
}
function mortgageReserveSpent(fromDate=null,toDate=null){
 return (data.regularPayments||[]).filter(p=>p.target==='mortgage'&&(!fromDate||p.date>=fromDate)&&(!toDate||p.date<=toDate))
  .reduce((sum,p)=>sum+num(p.reserveUsed===undefined?p.amount:p.reserveUsed),0);
}
function regularMortgageAvailable(date){
 if(!data.startBalances)return 0;
 return num(data.startBalances.mortgage)+data.receipts.filter(r=>r.date>=data.startBalances.date&&r.date<=date).reduce((a,r)=>a+num(r.amounts?.mortgage),0)+data.extraIncomes.filter(x=>x.destination==='mortgage'&&x.date>=data.startBalances.date&&x.date<=date).reduce((a,x)=>a+num(x.amount),0)-mortgageReserveSpent(data.startBalances.date,date);
}

// Fit the label and amount to the actual available tile width, including large balances.
function fitTotalMoneyTile(){
 const tile=document.querySelector('.hero-overview .hero-all');
 if(!tile)return;
 const amount=tile.querySelector('#heroAll');
 const label=tile.querySelector('span');
 if(!amount||!label)return;
 const available=tile.clientWidth-14;
 if(available<30)return;
 const canvas=document.createElement('canvas'),ctx=canvas.getContext('2d');
 for(const [el,max,min] of [[label,17,11],[amount,30,11]]){
  el.style.fontSize=max+'px';
  if(!ctx)continue;
  const computed=getComputedStyle(el);
  const family=computed.fontFamily||'sans-serif';
  const weight=computed.fontWeight||'700';
  let size=max;
  while(size>min){
   ctx.font=weight+' '+size+'px '+family;
   const width=ctx.measureText(el.textContent||'').width;
   if(width<=available)break;
   size-=.5;
  }
  el.style.fontSize=Math.max(min,size)+'px';
 }
}
