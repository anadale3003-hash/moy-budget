function undoReceipt(type){
 const key=month+'-'+type;
 const receipt=data.receipts.find(r=>r.id===key);
 if(!receipt){alert('Поступление не найдено.');return;}
 const title=type==='advance'?'аванс':'зарплату';
 if((num(receipt.amounts?.mortgage)>0&&regularMortgageAvailable(iso(new Date()))-num(receipt.amounts?.mortgage)<-0.001)||(num(receipt.amounts?.credit)>0&&regularCreditAvailable(iso(new Date()))-num(receipt.amounts?.credit)<-0.001)){alert('Нельзя отменить поступление: часть отложенных средств уже использована для подтверждённых платежей. Сначала отмени соответствующий платёж.');return;}
 if(!confirm('Отменить зачисление '+rub(receipt.amount)+' ('+title+') за '+month+'? Суммы будут вычтены из категорий. Уже записанные покупки и погашения сохранятся.'))return;
 data.receipts=data.receipts.filter(r=>r.id!==key);
 // Restore the planning amount too. Removing only the receipt used to leave
 // its large value in monthData(), forcing the salary forecast down to zero.
 const currentPlan=monthData();
 if(Math.abs(num(currentPlan[type])-num(receipt.amount))<0.001){
  currentPlan[type]=receipt.previousMonthAmount===undefined?0:num(receipt.previousMonthAmount);
 }
 // Only the credit-card allocation is also tracked in reserved; other allocations
 // are computed from receipts directly, so reversing them here would count twice.
 data.reserved=Math.max(0,num(data.reserved)-num(receipt.amounts?.credit));
 renderAll();
}

$('checkUpdate').addEventListener('click',async()=>{
  updateMessage('Проверяем новую версию…');
  if(!/^https:/.test(location.protocol)&&location.hostname!=='localhost'){
    updateMessage('Проверка работает после публикации приложения на HTTPS-сайте.');return;
  }
  try {
    const response=await fetch('./version.json?check='+Date.now(),{cache:'no-store'});
    if(!response.ok)throw new Error('HTTP '+response.status+' — файл version.json недоступен на сайте');
    const meta=await response.json();
    if(typeof meta.version!=='string')throw new Error('Некорректный формат version.json');
    const newer=(a,b)=>{const x=a.split('.').map(Number),y=b.split('.').map(Number);return x.some((n,i)=>n>(y[i]||0)&&x.slice(0,i).every((v,j)=>v===(y[j]||0)));};
    if(!newer(meta.version,APP_VERSION)){updateMessage('У тебя актуальная версия '+APP_VERSION);$('applyUpdate').style.display='none';return;}
    updateMessage('Доступна версия '+meta.version+'. Перед обновлением сохрани резервную копию.');
    $('applyUpdate').style.display='block';
    if('serviceWorker' in navigator){const reg=await navigator.serviceWorker.getRegistration();if(reg){await reg.update();pendingWorker=reg.waiting||reg.installing;}}
  }catch(err){updateMessage('Проверка не выполнена: '+(err?.message||'ошибка сети')+'. Проверь адрес /version.json на своём сайте.');}
});
$('applyUpdate').addEventListener('click',async()=>{
  updateMessage('Обновляем приложение…');
  if('serviceWorker' in navigator){
    const reg=await navigator.serviceWorker.getRegistration();
    if(reg){await reg.update();pendingWorker=reg.waiting||reg.installing;if(pendingWorker?.state==='installed')pendingWorker.postMessage({type:'SKIP_WAITING'});}
  }
  location.replace('./index.html?refresh='+Date.now());
});
