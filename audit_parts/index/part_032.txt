function paymentFunding(target,amount,date){
 const available=target==='mortgage'?regularMortgageAvailable(date):regularCreditAvailable(date);
 const reserve=Math.max(0,Math.round(available*100)/100);

function suggestedRegularPayment(target,date){

 // reserved as of the selected date, not the whole planned mortgage or debt.
 // Never suggest spending money from envelopes without a conscious choice.
 const already=regularPaidForMonth(target,month);
 const due=target==='mortgage'
  ?Math.max(0,num(monthData().mortgage)-already)
  :Math.max(0,num(data.debt)-repaymentTotal('credit')-regularPaymentSum('credit'));

}
