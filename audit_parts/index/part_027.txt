function savingsPlanRender(){
 const input=$('savingsPlanMonths');
 const parsed=Number(input.value);
 const validMonths=input.value.trim()!==''&&Number.isInteger(parsed)&&parsed>=1&&parsed<=120;
 const targetInput=$('savingsPlanTargetAmount');
 const proposedGoal=Number(targetInput.value);
 const reserveInput=$('ladderReserveInput');
 const reserveValue=Number(reserveInput.value);
 const incoming=reserveIncomeTotal();
 const validReserve=reserveInput.value.trim()!==''&&Number.isFinite(reserveValue)&&reserveValue>=incoming&&reserveValue<=1000000000;
 const reserve=validReserve?reserveValue:actualReserve();
 const savings=goalSavingsBalance();
 const booked=savings===null?0:Math.max(0,savings);
 const baseReserve=Math.max(0,reserve-incoming);
 const total=Math.round((reserve+($('ladderIncluded').checked?Math.max(0,booked-baseReserve):booked))*100)/100;
 const validGoal=targetInput.value.trim()!==''&&Number.isInteger(proposedGoal)
   &&proposedGoal>=reserve&&proposedGoal<=1000000000;
 const info={goal:proposedGoal,total,remaining:Math.max(0,proposedGoal-total),reserve};
 $('savingsPlanGoal').textContent=validGoal?rub(info.goal):'—';
 $('savingsPlanCurrent').textContent=rub(info.total);
 $('savingsPlanRemaining').textContent=validGoal?rub(info.remaining):'—';
 $('savingsPlanSave').disabled=!(validMonths&&validGoal&&validReserve);
 document.querySelectorAll('[data-savings-months]').forEach(btn=>{
   btn.classList.toggle('active',validMonths&&Number(btn.dataset.savingsMonths)===parsed);
 });
 if(!validMonths||!validGoal||!validReserve){
   $('savingsPlanPerMonth').textContent='—';
   $('savingsPlanDeadline').textContent=!validReserve?'Укажи корректный размер подушки':
    !validGoal?'Цель должна быть не ниже подушки: '+rub(info.reserve)
     :'Выбери срок от 1 до 120 месяцев';
   $('savingsPlanSplit').textContent='—';
   return;
 }
 const perMonth=Math.ceil(info.remaining/parsed);
 const fromAdvance=Math.floor(perMonth/2);
 const fromSalary=perMonth-fromAdvance;
 const deadline=savingsPlanTarget(parsed);
 const [y,m]=deadline.split('-').map(Number);
 const displayDate=new Date(y,m-1,1).toLocaleDateString('ru-RU',{month:'long',year:'numeric'});
 $('savingsPlanPerMonth').textContent=rub(perMonth);
 $('savingsPlanDeadline').textContent='Плановая цель — '+displayDate+' · '+parsed+' мес. включая текущий';
 $('savingsPlanSplit').textContent='Поровну: с аванса '+rub(fromAdvance)+' · с зарплаты '+rub(fromSalary);
}
