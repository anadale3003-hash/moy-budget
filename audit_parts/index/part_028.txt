function wifeFunds(date){
 // A confirmed balance is the source of truth. Earlier income, transfers
 // and reconciled expenses must never be counted again over that snapshot.
 const reports=(data.wifeBalanceReports||[])
  .filter(r=>r.date<=date&&(!data.startBalances||r.date>=data.startBalances.date)&&Number.isFinite(Number(r.family??r.total)))
  .sort((a,b)=>b.date.localeCompare(a.date)||String(b.checkedAt||'').localeCompare(String(a.checkedAt||'')));
 const last=reports[0];
 if(last){
  const cutoff=last.checkedAt||'';
  const afterSnapshot=entry=>entry.date<=date&&(entry.date>last.date||(entry.date===last.date&&!!entry.createdAt&&(!cutoff||entry.createdAt>cutoff)));
  const laterTransfers=(data.wifeMoves||[]).filter(afterSnapshot).reduce((sum,m)=>sum+num(m.amount),0);
  const laterPurchases=data.expenses.filter(e=>e.wifeCheck&&afterSnapshot(e)).reduce((sum,e)=>sum+num(e.amount),0);
  const laterWifeIncome=(data.extraIncomes||[]).filter(i=>(i.wifeIncome||i.source==='Доход жены')&&afterSnapshot(i)).reduce((sum,i)=>sum+num(i.amount),0);
  return Math.max(0,Math.round((num(last.family??last.total)+laterTransfers+laterWifeIncome-laterPurchases)*100)/100);
 }
 const moves=(data.wifeMoves||[]).filter(m=>m.date<=date&&(!data.startBalances||m.date>=data.startBalances.date));
 const sent=moves.reduce((sum,m)=>sum+num(m.amount),0);
 const food=data.extraIncomes.filter(i=>(i.supportTotal!=null||i.wifeIncome||i.source==='Доход жены')&&i.date<=date&&i.date>=data.startBalances?.date).reduce((sum,i)=>sum+num(i.amount),0);
 const spent=data.expenses.filter(e=>e.wifeCheck&&e.date<=date&&(!data.startBalances||e.date>=data.startBalances.date)).reduce((sum,e)=>sum+num(e.amount),0);
 const adjustments=(data.wifeAdjustments||[]).filter(a=>a.date<=date&&(!data.startBalances||a.date>=data.startBalances.date)).reduce((sum,a)=>sum+num(a.amount),0);
 return Math.max(0,Math.round((sent+food+adjustments-spent)*100)/100);
}
