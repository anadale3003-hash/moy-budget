/* v1.5.30: restore the portrait scroll position during reflow, without
   repeated delayed scroll resets that can produce a visible jump on iOS. */
(function(){
 const orientation=window.matchMedia('(orientation: landscape)');
 const app=document.querySelector('.app');
 const main=document.querySelector('.app>.main');
 const nav=document.querySelector('.app>.nav');
 let lastLandscape=orientation.matches;
 let settleUntil=0;
 let settleTimer=null;
 let framePending=false;

 function resetPortraitScroll(){
   if(orientation.matches)return;
   if(app && (app.scrollTop || app.scrollLeft)){app.scrollTop=0;app.scrollLeft=0}
   if(main && (main.scrollTop || main.scrollLeft)){main.scrollTop=0;main.scrollLeft=0}
   const root=document.scrollingElement;
   if(root && root.scrollTop)root.scrollTop=0;
   if(document.body && document.body.scrollTop)document.body.scrollTop=0;
   if(window.scrollY)window.scrollTo(0,0);
 }
 function settleOnNextFrame(){
   if(framePending)return;
   framePending=true;
   requestAnimationFrame(function(){
     framePending=false;
     resetPortraitScroll();
   });
 }
 function orientationChanged(){
   const landscape=orientation.matches;
   if(landscape!==lastLandscape){
     lastLandscape=landscape;
     if(settleTimer)clearTimeout(settleTimer);
     if(landscape){settleUntil=0;return}
     // Reset at the first possible opportunity, then only when Safari reports a resize.
     settleUntil=performance.now()+380;
     resetPortraitScroll();
     settleOnNextFrame();
     settleTimer=setTimeout(function(){settleUntil=0;},390);
   }else if(!landscape && settleUntil && performance.now()<settleUntil){
     settleOnNextFrame();
   }
 }
 if(nav)nav.addEventListener('click',function(event){
   if(orientation.matches && event.target.closest('[data-go]')){
     requestAnimationFrame(function(){
       if(app)app.scrollTop=0;
       if(main)main.scrollTop=0;
     });
   }
 });
 if(orientation.addEventListener)orientation.addEventListener('change',orientationChanged);
 else if(orientation.addListener)orientation.addListener(orientationChanged);
 window.addEventListener('orientationchange',orientationChanged,{passive:true});
 window.addEventListener('resize',orientationChanged,{passive:true});
 if(window.visualViewport)
   window.visualViewport.addEventListener('resize',orientationChanged,{passive:true});
})();
