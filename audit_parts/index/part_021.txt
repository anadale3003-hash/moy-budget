function nextPaydays(now=new Date(),pendingReceipt=null){
  // A salary paid on the 1st of the next calendar month belongs to the previous
  // planning cycle, whose advance was paid on the 16th. Keep calendar dates for
  // the credit deadline but associate both payments with the same plan month.
  const due=fromISO(data.deadline);if(!due)return [];
  const start=new Date(now.getFullYear(),now.getMonth(),now.getDate());
  const arr=[];
  const earlySalary=Number(data.payDays.salary)<=Number(data.payDays.advance);
  for(let i=-1;i<48;i++){
    const base=new Date(start.getFullYear(),start.getMonth()+i,1);
    const cycle=iso(base).slice(0,7);
    for(const type of ['advance','salary']){
      const shift=type==='salary'&&earlySalary?1:0;
      const target=new Date(base.getFullYear(),base.getMonth()+shift,1);
      const day=type==='advance'?data.payDays.advance:data.payDays.salary;
      const when=plannedPayday(iso(target).slice(0,7),Number(day)||1);
      if(when>=start&&when<=due&&!data.receipts.some(r=>r.id===cycle+'-'+type))arr.push({date:when,type,month:cycle});
    }
  }
  // The receipt being distributed already exists in data.receipts, so its
  // scheduled slot was skipped above. Count its actual payday exactly once.
  if(pendingReceipt&&start<=due){
    arr.push({date:start,type:pendingReceipt.type,month:pendingReceipt.month});
  }
  return arr.sort((a,b)=>a.date-b.date);
}
