function capitalSnapshot(goalOverride){
 const settings=ladderSettings(),reserve=actualReserve();
 const baseReserve=num(settings.reserve);
 const savings=goalSavingsBalance();
 const booked=savings===null?0:Math.max(0,savings);
 const extra=settings.included?Math.max(0,booked-baseReserve):booked;
 const total=Math.round((reserve+extra)*100)/100;
 const automaticGoal=total<1000000?1000000:(Math.floor(total/500000)+1)*500000;
 const userGoal=typeof settings.targetAmount==='number'&&Number.isFinite(settings.targetAmount)
   &&settings.targetAmount>=reserve&&settings.targetAmount<=1000000000;
 const override=typeof goalOverride==='number'&&Number.isInteger(goalOverride)
   &&goalOverride>=reserve&&goalOverride<=1000000000;
 const goal=override?goalOverride:(userGoal?settings.targetAmount:automaticGoal);
 return {settings,reserve,savings,extra,total,goal,automatic:!override&&!userGoal,
   remaining:Math.max(0,goal-total)};
}
function purposeBalances(snapshot=capitalSnapshot()){
 const cents=value=>Math.round(num(value)*100);
 let free=cents(snapshot.extra);
 const booked=new Map();
 const todayKey=iso(new Date());
 for(const rows of [data.receipts,data.extraIncomes,data.transfers]){
  for(const entry of rows||[]){
   if(entry.purposeId&&entry.date<=todayKey)
    booked.set(entry.purposeId,(booked.get(entry.purposeId)||0)+cents(entry.purposeAmount));
  }
 }
 const goals=(data.purposes||[]).map(goal=>{
  const assigned=Math.min(free,cents(goal.target),cents(goal.start)+(booked.get(goal.id)||0));
  free-=assigned;
  return {...goal,saved:assigned/100};
 });
 return {goals,free:free/100,active:goals.find(g=>g.id===data.activePurpose)};
}
function assignContribution(entry,amount){

 entry.purposeAmount=Math.min(amount,Math.max(0,Math.round((active.target-active.saved)*100)/100));
}

function nextCapitalMilestone(total,goal){
 return Math.min(goal,(Math.floor(total/10000)+1)*10000);
}
function milestoneText(total,goal){
 if(total>=goal)return 'Цель достигнута · '+rub(goal);
 const next=nextCapitalMilestone(total,goal);
 return 'До '+rub(next)+' осталось '+rub((Math.round(next*100)-Math.round(total*100))/100);
}
