/* Layout-only grouping. IDs/forms remain intact so accounting calculations and saved data are unchanged. */
(function(){
function group(section,title,nodes,extra){
 const outer=document.createElement('div');outer.className='design-group'+(extra?' '+extra:'');
 const heading=document.createElement('div');heading.className='section-heading';
 const h=document.createElement('h2');h.textContent=title;heading.appendChild(h);
 section.insertBefore(outer,nodes[0]);outer.appendChild(heading);
 nodes.forEach(n=>outer.appendChild(n));return outer;
}
function groupHeadings(id){
 const section=document.getElementById(id);if(!section)return;
 const children=Array.from(section.children);let start=-1;
 for(let i=0;i<children.length;i++){
  if(children[i].classList.contains('section-heading')){
   if(start>=0){const nodes=children.slice(start,i);if(nodes.length>1)groupExisting(section,nodes)}
   start=i;
  }
 }
 if(start>=0){const nodes=children.slice(start);if(nodes.length>1)groupExisting(section,nodes)}
}
function groupExisting(section,nodes){
 const outer=document.createElement('div');outer.className='design-group';
 section.insertBefore(outer,nodes[0]);nodes.forEach(n=>outer.appendChild(n));
}
function setup(){
 const section=document.getElementById('v-card');
 if(section){
  Array.from(section.querySelectorAll(':scope > .section-heading')).forEach(h=>h.remove());
  const pay=section.querySelector('.payment-obligations');
  const mortgage=section.querySelector('.mortgage-monthly-settings');
  const credit=pay?.nextElementSibling?.classList.contains('card')?pay.nextElementSibling:null;
  // Credit form follows its heading in the original source.
  const creditField=document.getElementById('debt');
  const creditCard=creditField?.closest('.card');
  if(pay)section.insertBefore(pay,section.firstElementChild);
  if(pay)group(section,'Фактическая оплата',[pay],'credit-payments-group');
  if(mortgage)group(section,'Ипотека',[mortgage],'credit-mortgage-group');
  if(creditCard){creditCard.classList.add('credit-basic-card');group(section,'Кредитка',[creditCard],'credit-balance-group')}
  const all=Array.from(section.children);
  all.filter(n=>!n.classList.contains('design-group')).forEach(n=>n.classList.add('legacy-credit-extras'));
 }
 ['v-plan','v-expense','v-analytics','v-settings'].forEach(groupHeadings);
 // Compress only the payouts section: advanced weights stay available in a closed disclosure.
 const plan=document.getElementById('v-plan');
 const distribution=plan?.querySelector('.payout-destinations');
 const weightCard=plan?.querySelector('#weight-family')?.closest('.card');
 if(distribution&&weightCard){
   const fold=document.createElement('details');fold.className='compact-disclosure plan-weights-disclosure';
   const summary=document.createElement('summary');summary.textContent='Пропорции остатка';
   fold.appendChild(summary);distribution.parentElement.insertBefore(fold,weightCard);fold.appendChild(weightCard);
 }
 const history=document.getElementById('regularPaymentHistory');
 if(history){
   const fold=document.createElement('details');fold.className='compact-disclosure credit-history-disclosure';
   const summary=document.createElement('summary');summary.textContent='История оплат';
   history.parentElement.insertBefore(fold,history);fold.appendChild(summary);fold.appendChild(history);
 }
 // Separate history, charts and settings into individually framed groups.
 function separateItems(id){
  const section=document.getElementById(id),first=section?.querySelector(':scope > .design-group');
  if(!first)return;
  const content=Array.from(first.children);
  const items=content.filter(el=>el.classList.contains('card')||el.matches('details.compact-disclosure'));
  let insertionPoint=first;
  items.slice(1).forEach((el,i)=>{
   const wrap=document.createElement('div');wrap.className='design-group';
   const isDetails=el.matches('details');
   if(!isDetails){
    const title=id==='v-analytics'?(i===0?'Поступления':i===1?'Основные категории':'Виды расходов'):
      (el.querySelector('.row>b')?.textContent||'Параметры');
    const heading=document.createElement('div');heading.className='section-heading';
    const h=document.createElement('h2');h.textContent=title;heading.appendChild(h);wrap.appendChild(heading);
    if(id==='v-settings'){const b=el.querySelector('.row>b');if(b)b.remove()}
    if(id==='v-analytics'&&i>=1){const b=el.querySelector('.row>b');if(b)b.remove()}
   }
   first.parentElement.insertBefore(wrap,insertionPoint.nextElementSibling);
   wrap.appendChild(el);insertionPoint=wrap;
  });
 }
 ['v-expense','v-analytics','v-settings'].forEach(separateItems);
}
if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',setup,{once:true});
else setup();
})();
