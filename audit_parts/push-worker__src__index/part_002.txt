  async scheduled(_event, env, ctx) {
    ctx.waitUntil((async () => {
      const now = new Date(_event?.scheduledTime || Date.now());
      const today = moscowDate(now);
      // The Moscow timezone is UTC+03:00 year-round.
      const hour = String((now.getUTCHours() + 3) % 24).padStart(2, "0");
      const records = await storeCall(env, "/list");
      for (const record of records) {
        const prefs = record.preferences || {};
        if (prefs.reminders === "none") continue;
        const due = duePaydays(today, prefs).filter(item =>
          !record.confirmed?.[item.period + "-" + item.type]
        );
        const messages = due.map(item => ({ type: item.type, period: item.period }));
        if (prefs.reminders === "all" && isMonthEnd(today)) messages.push({ type: "month-end", period: today.slice(0, 7) });
        for (const item of messages) {
          const type = item.type;
          // Salary and advance reminders have a per-HOUR key. Month-end
          // analytics remains once daily to avoid 13 identical summaries.
          const sentKey = today + ":" + item.period + ":" + type +
            (type === "month-end" ? ":daily" : ":" + hour);
          const confirmKey = type === "month-end" ? null : item.period + "-" + type;
          // Atomic claim in Durable Object prevents overlapping Cron invocations
          // from sending a duplicate. Confirmations are rechecked inside the claim.
          const claim = await storeCall(env, "/claim", { id: record.id, sentKey, confirmKey });
          if (!claim.ok) continue;
          try {
            const message = notificationFor(type);
            if (type !== "month-end") message.tag = type + "-" + today + "-" + hour;
            const response = await sendToRecord(env, record, message);
            if (response.status === 404 || response.status === 410) {
              await storeCall(env, "/delete", { id: record.id });
              break;
            }
            if (response.ok) {
              await storeCall(env, "/mark-sent", { id: record.id, sentKey });
            } else {
              console.warn("Push rejected", response.status, "subscription", record.id.slice(0, 8));
              await storeCall(env, "/release-claim", { id: record.id, sentKey });
            }
          } catch (error) {
            console.error("Scheduled push failed", String(error?.message || error));
            await storeCall(env, "/release-claim", { id: record.id, sentKey });
          }
        }
      }
    })());
  }
